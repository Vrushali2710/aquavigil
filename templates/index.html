Distributed Spatial Join Statement: 
WITH DepressurizedWaterSegments AS (
    SELECT w.pipe_id, w.sector_id, w.geometry AS water_geom, 
           ST_BUFFER(w.geometry, 5.0) AS vulnerability_buffer
    FROM `aquavigil.water_network` w 
    JOIN `aquavigil.scada_valve_events` v ON w.valve_id = v.valve_id 
    WHERE v.event_type = 'SUPPLY_SHUTDOWN' 
      AND v.timestamp >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 2 HOUR)
)
SELECT ws.pipe_id, s.sewer_id,
       ROUND(ST_DISTANCE(ws.water_geom, s.geometry), 2) AS proximity_m,
       ROUND(((5.0 - ST_DISTANCE(ws.water_geom, s.geometry))/5.0)*0.45 
             + (s.condition_rating/5.0)*0.35 + 0.20, 2) AS intrusion_risk_score,
       ST_ASGEOJSON(ST_INTERSECTION(ws.vulnerability_buffer, s.geometry)) AS corridor_geojson
FROM DepressurizedWaterSegments ws 
JOIN `aquavigil.sewerage_network` s 
  ON ST_INTERSECTS(ws.vulnerability_buffer, s.geometry)
ORDER BY intrusion_risk_score DESC;
